-- Practice adds immutable assignment integrity and submission snapshots without
-- guessing bindings for any pre-existing incomplete baseline session.
ALTER TABLE practice_sessions ADD COLUMN practice_sha256 TEXT CHECK(practice_sha256 IS NULL OR length(practice_sha256)=64);
ALTER TABLE practice_sessions ADD COLUMN snapshot_sha256 TEXT CHECK(snapshot_sha256 IS NULL OR length(snapshot_sha256)=64);
ALTER TABLE practice_sessions ADD COLUMN submission_json TEXT CHECK(submission_json IS NULL OR json_valid(submission_json));
ALTER TABLE practice_sessions ADD COLUMN submission_sha256 TEXT CHECK(submission_sha256 IS NULL OR length(submission_sha256)=64);
CREATE INDEX practice_session_workspace ON practice_sessions(workspace_id,id);
CREATE TRIGGER immutable_practice_assignment BEFORE UPDATE OF workspace_id,practice_id,practice_revision,question_refs_json,solution_refs_json,practice_sha256,snapshot_sha256 ON practice_sessions WHEN OLD.workspace_id IS NOT NEW.workspace_id OR OLD.practice_id IS NOT NEW.practice_id OR OLD.practice_revision IS NOT NEW.practice_revision OR OLD.question_refs_json IS NOT NEW.question_refs_json OR OLD.solution_refs_json IS NOT NEW.solution_refs_json OR OLD.practice_sha256 IS NOT NEW.practice_sha256 OR OLD.snapshot_sha256 IS NOT NEW.snapshot_sha256 BEGIN SELECT RAISE(ABORT,'practice assignment immutable'); END;
CREATE TRIGGER immutable_practice_submission BEFORE UPDATE OF submission_json,submission_sha256 ON practice_sessions WHEN OLD.submission_json IS NOT NULL AND (OLD.submission_json IS NOT NEW.submission_json OR OLD.submission_sha256 IS NOT NEW.submission_sha256) BEGIN SELECT RAISE(ABORT,'practice submission immutable'); END;
CREATE TRIGGER practice_terminal_not_active BEFORE UPDATE OF status ON practice_sessions WHEN OLD.status<>'active' AND NEW.status<>OLD.status BEGIN SELECT RAISE(ABORT,'practice terminal status immutable'); END;
CREATE TRIGGER practice_response_insert_active BEFORE INSERT ON practice_responses WHEN NOT EXISTS(SELECT 1 FROM practice_sessions WHERE id=NEW.session_id AND status='active') BEGIN SELECT RAISE(ABORT,'practice session not active'); END;
CREATE TRIGGER practice_response_update_active BEFORE UPDATE ON practice_responses WHEN NOT EXISTS(SELECT 1 FROM practice_sessions WHERE id=OLD.session_id AND status='active') BEGIN SELECT RAISE(ABORT,'practice session not active'); END;
CREATE TRIGGER practice_response_delete_active BEFORE DELETE ON practice_responses WHEN NOT EXISTS(SELECT 1 FROM practice_sessions WHERE id=OLD.session_id AND status='active') BEGIN SELECT RAISE(ABORT,'practice session not active'); END;
CREATE TABLE practice_exposures(session_id TEXT NOT NULL REFERENCES practice_sessions(id),question_id TEXT NOT NULL,kind TEXT NOT NULL CHECK(kind IN ('hint','solution')),level INTEGER NOT NULL CHECK((kind='hint' AND level BETWEEN 1 AND 3) OR (kind='solution' AND level=0)),exposure_id TEXT NOT NULL UNIQUE REFERENCES exposures(id),help_json TEXT NOT NULL CHECK(json_valid(help_json)),help_sha256 TEXT NOT NULL CHECK(length(help_sha256)=64),PRIMARY KEY(session_id,question_id,kind,level));
CREATE TRIGGER practice_help_no_update BEFORE UPDATE ON practice_exposures BEGIN SELECT RAISE(ABORT,'practice help history immutable'); END;
