TEXT = '独立审查 🧠 e\u0301 合成正文。\n'


def tree(rev=1,suffix=''):
    raw=(TEXT+f'修订 {rev}\n').encode()
    block=dm.ContentBlock(id='review_block'+suffix,revision=rev,kind='text',title='合成块'+suffix,
                          body_path=f'content/body{suffix}-{rev}.md',body_sha256=sha256(raw).hexdigest())
    lesson=dm.Lesson(id='review_lesson'+suffix,revision=rev,title='合成小节',objectives=[],block_refs=[reference(block)])
    course=dm.Course(id='review_course'+suffix,revision=rev,title='合成课程',audience='review',lesson_refs=[reference(lesson)])
    return [block,lesson,course],{block.body_path:raw}


def setup(base):
    db=Database(Settings(data_dir=base/'data'))
    workspace=db.initialize()
    actor=SessionIdentity('review_session',workspace,'learner','synthetic','2099-01-01T00:00:00Z')
    values,bodies=tree()
    ContentService(db).publish(workspace,values,bodies)
    return db,actor,values,bodies


def note(actor,block,name='review_note'):
    start=TEXT.index('🧠')
    return dm.Note(id=name,revision=1,workspace_id=actor.workspace_id,markdown='合成笔记',
        anchor=dm.Selection(ref=reference(block),exact_quote='🧠',prefix=TEXT[:start],suffix=TEXT[start+1:],
                            start_codepoint=start,end_codepoint=start+1))


def count(db):
    with db.connect() as connection:
        return {table:connection.execute('SELECT COUNT(*) FROM '+table).fetchone()[0]
                for table in ['objects','revisions','notes_index','learning_events','learning_progress','idempotency','outbox','evidence']}


def concurrent_cas(base):
    db,actor,values,_=setup(base)
    notes=NotesService(db)
    original=note(actor,values[0])
    oldref=notes.create(actor,original,'create')
    barrier=Barrier(2)
    candidates=[original.model_copy(update={'revision':2,'markdown':'candidate_'+name}) for name in ['a','b']]
    def apply(pair):
        index,candidate=pair
        barrier.wait(timeout=5)
        try:
            result=NotesService(Database(db.settings)).update(actor,original.id,candidate,oldref.sha256,'update_'+str(index))
            return {'index':index,'status':200,'ref':result.model_dump(mode='json')}
        except ApiError as exc:
            return {'index':index,'status':exc.status,'code':exc.code}
    with ThreadPoolExecutor(max_workers=2) as executor:
        outcomes=list(executor.map(apply,enumerate(candidates)))
    assert sorted(item['status'] for item in outcomes)==[200,412],outcomes
    winner=next(item for item in outcomes if item['status']==200)
    assert notes.list(actor.workspace_id).items[0] == candidates[winner['index']]
    assert count(db)['learning_events']==1
    with db.connect() as connection:
        stored=ContentRepository(connection,actor.workspace_id,allow_notes=True).load('note',original.id,1).value
        assert stored == original and reference(stored)==oldref
    new,bodies=tree(2)
    ContentService(db).publish(actor.workspace_id,new,bodies)
    stale=notes.list(actor.workspace_id).items[0]
    assert stale.revision==3 and stale.anchor_state=='stale' and stale.anchor==original.anchor
    replay=notes.update(actor,original.id,candidates[winner['index']],oldref.sha256,'update_'+str(winner['index']))
    assert replay.model_dump(mode='json')==winner['ref']
    assert notes.list(actor.workspace_id).items[0]==stale
    return {'status':'PASS','competing_statuses':sorted(item['status'] for item in outcomes),
            'current_note_revision_after_publication':stale.revision,'old_hash_unchanged':True,
            'replay_returns_original_receipt_without_overwrite':True}


def batch_rollback(base):
    db,actor,values,_=setup(base)
    notes=NotesService(db)
    original=note(actor,values[0])
    notes.create(actor,original,'create')
    other,other_bodies=tree(suffix='_other')
    ContentService(db).publish(actor.workspace_id,other,other_bodies)
    unrelated=note(actor,other[0],'review_unrelated')
    notes.create(actor,unrelated,'other_note')
    v2,b2=tree(2)
    v3,b3=tree(3)
    before=count(db)
    with db.transaction() as connection:
        connection.execute("CREATE TRIGGER review_fail_stale BEFORE INSERT ON outbox WHEN NEW.event_type='note.anchor_stale' BEGIN SELECT RAISE(ABORT,'synthetic review failure'); END")
    try:
        with db.transaction() as connection:
            ContentService(db).publish_in_transaction(connection,actor.workspace_id,[*v3,*v2],{**b2,**b3},import_history=True)
    except Exception as error:
        assert 'synthetic review failure' in str(error) or isinstance(error,ApiError)
    else:
        raise AssertionError('Injected stale failure did not abort publication')
    assert count(db)==before
    assert ContentService(db).current(actor.workspace_id,values[0].id)==reference(values[0])
    assert notes.list(actor.workspace_id).items == [original,unrelated]
    with db.transaction() as connection:
        connection.execute('DROP TRIGGER review_fail_stale')
        ContentService(db).publish_in_transaction(connection,actor.workspace_id,[*v3,*v2],{**b2,**b3},import_history=True)
    current={item.id:item for item in notes.list(actor.workspace_id).items}
    assert current[original.id].revision==2 and current[original.id].anchor_state=='stale'
    assert current[unrelated.id]==unrelated
    assert ContentService(db).current(actor.workspace_id,values[0].id)==reference(v3[0])
    with db.connect() as connection:
        assert connection.execute("SELECT COUNT(*) FROM outbox WHERE event_type='note.anchor_stale'").fetchone()[0]==1
        assert ContentRepository(connection,actor.workspace_id,allow_notes=True).load('note',original.id,1).value==original
    return {'status':'PASS','all_sql_counts_rollback':True,'batch_current_block_revision':3,
            'new_stale_note_revisions':1,'unrelated_note_unchanged':True}


def guard_http(base):
    db,actor,values,_=setup(base)
    app=create_app(db.settings)
    with TestClient(app,base_url=db.settings.origin) as client:
        bootstrap=client.post('/api/v1/session/bootstrap',json={'one_time_code':issue_bootstrap_code(db)},headers={'Origin':db.settings.origin})
        assert bootstrap.status_code==200
        headers={'Origin':db.settings.origin,'X-CSRF-Token':bootstrap.json()['csrf_token'],'Idempotency-Key':'guard_create'}
        before=count(db)
        for _ in range(4):
            progress=client.get('/api/v1/learning/progress')
            assert progress.status_code==200 and progress.json()=={'revision':1,'readings':[],'route_steps':[],'bookmarks':[]}
        assert count(db)==before
        original=note(actor,values[0])
        saved=client.post('/api/v1/notes',json=original.model_dump(mode='json'),headers=headers)
        assert saved.status_code==201
        with db.transaction() as connection:
            connection.execute("INSERT INTO workspace(id,title,created_at) VALUES('foreign_review','合成','2026-09-14T00:00:00Z')")
        foreign=replace(actor,workspace_id='foreign_review')
        assert NotesService(db).list(foreign.workspace_id).items==[]
        try:
            NotesService(db).create(foreign,original.model_copy(update={'workspace_id':foreign.workspace_id,'id':'foreign_note'}),'foreign_create')
        except ApiError as error:
            assert error.status==404
        else:
            raise AssertionError('Cross-workspace anchor was accepted')
        # Insert a genuine published assessment plus an independent guard row.
        concept=dm.Concept(id='review_concept',revision=1,title='合成概念')
        question=dm.QuestionPublic(id='review_question',revision=1,kind='numeric',stem_markdown='合成题',concept_ids=[concept.id],skill='compute',exposure_group='review_group',input_instructions='数字')
        assessment=dm.AssessmentBlueprint(id='review_assessment',revision=1,title='合成测试',question_refs=[reference(question)],allowed_modes=['independent'])
        ContentService(db).publish(actor.workspace_id,[concept,question,assessment],{})
        with db.transaction() as connection:
            connection.execute("INSERT INTO attempts(id,workspace_id,assessment_id,assessment_revision,mode,policy_json,question_refs_json,solution_refs_private_json,status,revision,created_at) VALUES('review_attempt',?,'review_assessment',1,'independent','{}','[]','[]','active',1,'2026-09-14T00:00:00Z')",(actor.workspace_id,))
        guarded_count=count(db)
        responses=[]
        for role in ['learner','author']:
            with db.transaction() as connection:
                connection.execute('UPDATE local_sessions SET role=? WHERE workspace_id=?',(role,actor.workspace_id))
            responses.append(client.post('/api/v1/notes',json=original.model_dump(mode='json'),headers=headers).status_code)
            responses.append(client.get('/api/v1/notes').status_code)
            responses.append(client.get('/api/v1/objects/'+original.id+'/current').status_code)
            responses.append(client.get('/api/v1/objects/'+original.id+'/revisions').status_code)
            responses.append(client.get('/api/v1/learning/progress').status_code)
        assert set(responses)=={409},responses
        assert count(db)==guarded_count
    return {'status':'PASS','empty_gets_created_events':0,'foreign_anchor_status':404,
            'learner_author_guarded_responses':responses,'successful_command_replay_guarded':True}
