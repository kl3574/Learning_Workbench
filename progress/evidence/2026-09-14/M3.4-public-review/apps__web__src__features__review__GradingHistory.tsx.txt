import { useEffect, useRef, useState } from 'react'
import type { GradeHistoryEntry } from '../../../../../packages/contracts/generated/api-types'
import { readLocal, writeLocal } from '../../workbench/uiCache'
import { scoreText } from './reviewModel'
import { EvidenceFacts } from './EvidenceFacts'
export function GradingHistory({ workspace, tabId, history, questionId, selected }: { workspace: string; tabId: string; history: GradeHistoryEntry[]; questionId: string | null; selected: (value: GradeHistoryEntry | null) => void }) {
  const key = `learning-workbench.${workspace}.${tabId}.review-grade.v1`
  const [wanted, setWanted] = useState<number | null>(() => { const value = readLocal(key); return value === null ? null : /^[1-9][0-9]*$/.test(value) && Number.isSafeInteger(Number(value)) ? Number(value) : -1 })
  const [error, setError] = useState(''), [comparison, setComparison] = useState<number | null>(null)
  const entry = history.find(value => value.grading_revision === (wanted ?? history.at(-1)?.grading_revision)) ?? null
  const item = entry?.items.find(value => value.question_ref.id === questionId)
  const compare = history.find(value => value.grading_revision === comparison) ?? null
  const callback = useRef(selected); callback.current = selected
  useEffect(() => { callback.current(entry) }, [entry])
  const select = (revision: number) => { setWanted(revision); setError(writeLocal(key, String(revision)) ? '' : '所选评分版本尚未保存到本机；当前选择仍保留。') }
  return <section className="review-history" aria-label="完整评分历史"><h2>评分版本与证据</h2><p>服务端保留 {history.length} 个已完成评分版本。历史摘要只包含分数、精确映射与资格，不提供历史反馈或标准答案。</p>{history.length > 0 ? <><label>选择评分版本<select data-focus-key="review-grade-version" aria-label="选择评分版本" value={entry?.grading_revision ?? ''} onChange={event => select(Number(event.target.value))}>{!entry && <option value="">保存的版本当前不可读</option>}{history.map(value => <option key={value.grading_revision} value={value.grading_revision}>评分版本 {value.grading_revision} · {value.status === 'needs_review' ? '仍有待复核项' : '已评分'} · {value.finalized_at}</option>)}</select></label>{!entry && <p role="alert">保存的评分版本 {wanted === -1 ? '格式无效' : wanted} 不在本次完整回读中。未自动换成最新版本；请明确选择可用版本。</p>}{entry && <><p>正在查看评分版本 {entry.grading_revision}。{entry.grading_revision === history.at(-1)?.grading_revision ? '这是最近完成的评分；当前任务状态仍以上方回执为准。' : '这是历史评分摘要，不是当前评分结果。'}原评分时间：<time dateTime={entry.finalized_at}>{entry.finalized_at}</time>；规则 {entry.grading_rules_version}。</p>{item && <><p className="review-selected-score">本题在版本 {entry.grading_revision}：{scoreText(item)}</p><EvidenceFacts item={item} entry={entry} /></>}<details data-view-key="review-grade-comparison"><summary data-focus-key="review-grade-comparison">比较评分版本</summary><label>对照版本<select aria-label="对照评分版本" value={comparison ?? ''} onChange={event => setComparison(event.target.value ? Number(event.target.value) : null)}><option value="">请选择另一已完成版本</option>{history.filter(value => value.grading_revision !== entry.grading_revision).map(value => <option key={value.grading_revision} value={value.grading_revision}>评分版本 {value.grading_revision}</option>)}</select></label>{compare && compare.grading_revision !== entry.grading_revision && <div className="table-scroll" tabIndex={0} aria-label="逐题评分版本对照"><table><thead><tr><th>题目</th><th>版本 {compare.grading_revision}</th><th>版本 {entry.grading_revision}</th></tr></thead><tbody>{entry.items.map((value, index) => <tr key={value.question_ref.id}><th>第 {index + 1} 题</th><td>{scoreText(compare.items[index])}<br />{compare.items[index].eligible ? '纳入证据' : '保留记录，不纳入证据'}</td><td>{scoreText(value)}<br />{value.eligible ? '纳入证据' : '保留记录，不纳入证据'}</td></tr>)}</tbody></table></div>}</details></>}</> : <p>尚无已完成评分版本。未完成的评分任务不会生成占位成绩。</p>}{error && <p role="alert">{error}</p>}</section>
}
