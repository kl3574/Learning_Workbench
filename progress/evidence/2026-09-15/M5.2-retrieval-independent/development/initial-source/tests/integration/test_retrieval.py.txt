def all_rows(database):
    with database.connect() as connection:
        names = [row[0] for row in connection.execute("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
        return {name: sorted([tuple(row) for row in connection.execute(f'SELECT * FROM "{name}"')], key=repr)
                for name in names}

def request_for(service, identity, refs):
    state = service.scope_status(identity, refs)
    return RetrievalIndexRebuildWrite(scope_refs=refs, expected_corpus_sha256=state.corpus_sha256,
                                     provider_id=None, consent_id=None)


def built(retrieval, refs=None, key='build'):
    database, identity, _, course, service, worker = retrieval
    refs = refs or [reference(course)]
    request = request_for(service, identity, refs)
    ack = service.rebuild(identity, request, key)
    for _ in range(5):
        current = service.job(identity, ack.id)
        if current.status in {'completed', 'failed', 'cancelled'}:
            break
        worker.run_once()
    assert service.job(identity, ack.id).status == 'completed'
    state = service.scope_status(identity, refs)
    assert state.state == 'ready'
    return request, ack, state


def query(retrieval, text='SVD', refs=None, limit=5):
    _, identity, _, course, service, _ = retrieval
    return service.query(identity, RetrievalQueryWrite(query=text, scope_refs=refs or [reference(course)], limit=limit))


def publish_small(database, identity, prefix, texts, *, title_size=0):
    blocks = []
    bodies = {}
    for number, text in enumerate(texts):
        body = text.encode()
        block = dm.ContentBlock(id=f'{prefix}_block_{number:03d}', revision=1, kind='text',
            title='T' * title_size if title_size else f'公开材料 {number}',
            body_path=f'content/{prefix}_{number:03d}.md', body_sha256=hashlib.sha256(body).hexdigest())
        blocks.append(block)
        bodies[block.body_path] = body
    lesson = dm.Lesson(id=f'{prefix}_lesson', revision=1, title='L' * title_size if title_size else '公开小节',
        objectives=[], block_refs=[reference(block) for block in blocks])
    course = dm.Course(id=f'{prefix}_course', revision=1, title='C' * title_size if title_size else '明确范围',
        audience='软件测试未审材料', lesson_refs=[reference(lesson)])
    ContentService(database).publish(identity.workspace_id, (*blocks, lesson, course), bodies)
    return blocks, lesson, course

    assert ack.status == 'queued' and service.job(identity, ack.id).status == 'completed'
    assert service.scope_status(identity, [reference(course)]).index_version == state.index_version
    assert all_rows(database) == before

@pytest.mark.parametrize('damage', ['body', 'manifest', 'chunk', 'fts', 'fts_scope'])
def test_actual_body_manifest_chunk_or_fts_damage_is_never_successful_empty(retrieval, damage):
    database, _, blocks, _, _, _ = retrieval
    built(retrieval)
    if damage == 'body':
        digest = blocks[19].body_sha256
        path = database.settings.data_dir / 'blobs' / digest[:2] / digest
        path.write_bytes(b'tampered public body')
    else:
        # Deliberately damage only this test database, bypassing immutable
        # triggers. These are corruption fixtures, not product edit entry points.
        with database.transaction() as connection:
            if damage == 'manifest':
                connection.execute('DROP TRIGGER retrieval_generations_no_update')
                connection.execute('UPDATE retrieval_generations SET manifest_sha256=?', ('0' * 64,))
            elif damage == 'chunk':
                connection.execute('DROP TRIGGER retrieval_chunks_no_update')
                connection.execute("UPDATE retrieval_chunks SET tokens='w666f72676564'")
            elif damage == 'fts':
                connection.execute("UPDATE retrieval_fts SET tokens='w666f72676564'")
            else:
                connection.execute('UPDATE retrieval_fts SET scope_sha256=?', ('0' * 64,))
    with pytest.raises(ApiError) as failure:
        query(retrieval)
    assert failure.value.code == ('CONTENT_HASH_MISMATCH' if damage == 'body' else 'INDEX_INTEGRITY_INVALID')

@pytest.mark.parametrize('running', [False, True])
def test_public_jobs_cancel_queued_or_running_never_publishes_a_generation(retrieval, running):
    database, identity, _, course, service, worker = retrieval
    fixture, assessments = assessment_material(database, identity)
    request = request_for(service, identity, [reference(course)])
    ack = service.rebuild(identity, request, 'original')
    lease = worker.claim() if running else None
    computed = worker.compute(lease) if lease else None
    jobs = JobService(database)
    current = jobs.job(identity, ack.id)
    assert current.status == ('running' if running else 'queued')
    with pytest.raises(ApiError) as excluded:
        assessments.create_attempt(identity, fixture.assessment.id,
            AssessmentAttemptCreate(assessment_ref=reference(fixture.assessment), mode='independent'), 'blocked')
    assert excluded.value.code == 'SUBJECT_WORK_ACTIVE'
    cancel = JobCancelRequest(expected_revision=current.revision)
    original = jobs.cancel(identity, ack.id, cancel, 'cancel')
    assert original.status == 'cancelled' and original.result_refs == []
    assert jobs.cancel(identity, ack.id, cancel, 'cancel') == original
    if computed:
        assert not worker.finish(lease, *computed)
    worker.run_once()  # May consume a real maintenance event; not evidence of publication.
    assert service.scope_status(identity, [reference(course)]).state == 'missing'
    assert jobs.job(identity, ack.id).status == 'cancelled'
    with database.connect() as connection:
        assert connection.execute('SELECT COUNT(*) FROM retrieval_generations').fetchone()[0] == 0
        assert connection.execute("SELECT COUNT(*) FROM job_events WHERE job_id=? AND type IN ('completed','failed','cancelled')", (ack.id,)).fetchone()[0] == 1
    assert assessments.create_attempt(identity, fixture.assessment.id,
        AssessmentAttemptCreate(assessment_ref=reference(fixture.assessment), mode='independent'), 'after-cancel').status == 'active'


def test_expired_lease_recovery_excludes_old_writer_and_commits_one_generation(retrieval):
    from services.api.app.application.retrieval import RetrievalWorker

    database, identity, _, course, service, worker = retrieval
    ack = service.rebuild(identity, request_for(service, identity, [reference(course)]), 'build')
    first = worker.claim()
    assert first is not None
    old_material = worker.compute(first)
    with database.transaction() as connection:
        connection.execute('UPDATE jobs SET lease_until=? WHERE id=?', ('2000-01-01T00:00:00Z', ack.id))
    restarted = RetrievalWorker(Database(database.settings))
    second = restarted.claim()
    assert second and second.job_id == first.job_id and second.owner != first.owner and second.revision > first.revision
    new_material = restarted.compute(second)
    assert worker.finish(first, *old_material) is False
    assert restarted.finish(second, *new_material) is True
    assert service.job(identity, ack.id).status == 'completed'
    with database.connect() as connection:
        assert connection.execute('SELECT COUNT(*) FROM retrieval_generations').fetchone()[0] == 1
        assert connection.execute("SELECT COUNT(*) FROM job_events WHERE job_id=? AND type='completed'", (ack.id,)).fetchone()[0] == 1

def test_stop_after_compute_prevents_publication_and_allows_lease_recovery(retrieval):
    from services.api.app.application.retrieval import RetrievalWorker

    database, identity, _, course, service, _ = retrieval
    stop = threading.Event()
    worker = RetrievalWorker(database, stopping=stop.is_set)
    ack = service.rebuild(identity, request_for(service, identity, [reference(course)]), 'build')
    lease = worker.claim()
    assert lease is not None
    material = worker.compute(lease)
    stop.set()
    assert not worker.finish(lease, *material)
    assert service.job(identity, ack.id).status == 'running'
    assert service.scope_status(identity, [reference(course)]).index_version is None
    with database.transaction() as connection:
        connection.execute('UPDATE jobs SET lease_until=? WHERE id=?', ('2000-01-01T00:00:00Z', ack.id))
    resumed = RetrievalWorker(Database(database.settings))
    assert resumed.run_once()
    assert service.job(identity, ack.id).status == 'completed'

@pytest.mark.parametrize('kind', ['text', 'json'])
def test_actual_full_block_resource_omissions_preserve_text_and_all_parent_fields(retrieval, kind):
    database, identity, _, _, _, _ = retrieval
    body = 'needle ' + 'x ' * 199990 if kind == 'text' else 'needle'
    count = 6 if kind == 'text' else 8
    blocks, _, course = publish_small(database, identity, f'budget_{kind}', [body] * count,
                                      title_size=400000 if kind == 'json' else 0)
    refs = [reference(course)]
    built(retrieval, refs)
    result = query(retrieval, 'needle', refs, limit=20)
    assert result.result_state == 'matched' and result.matched_count == count
    assert 0 < len(result.hits) < count
    assert result.omissions.result_limit == 0
    assert result.omissions.text_byte_budget + result.omissions.json_byte_budget == count - len(result.hits)
    assert (result.omissions.text_byte_budget > 0) == (kind == 'text')
    assert (result.omissions.json_byte_budget > 0) == (kind == 'json')
    assert sum(len(hit.text.encode()) for hit in result.hits) <= 2 * 1024 * 1024
    assert len(result.model_dump_json().encode()) <= 8 * 1024 * 1024
    assert all(hit.text == body and hit.location.end_cp == len(body) for hit in result.hits)
    if kind == 'json':
        assert all(len(hit.title) == len(hit.parent_paths[0].lesson_title) == len(hit.parent_paths[0].course_title) == 400000 for hit in result.hits)
    assert {hit.ref.id for hit in result.hits} <= {block.id for block in blocks}
