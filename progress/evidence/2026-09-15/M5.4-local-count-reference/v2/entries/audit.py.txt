from pathlib import Path

A=Path('/audit'); O=Path('/output'); T=Path('/materials/static/tokenizers/v41/tokenizer.json')
TSHA='81f64d1248a68ce3663e07ab3ee48b851e5df0e32d27cb98e4c9a268151e8d99'
def sha(b):return hashlib.sha256(b).hexdigest()
def canon(v):return json.dumps(v,ensure_ascii=False,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
def pairs(p):
 d={}
 for k,v in p:
  if k in d:raise ValueError('duplicate field')
  d[k]=v
 return d
def invalid(_):raise ValueError('nonfinite')
def admit(raw):
 v=json.loads(raw,object_pairs_hook=pairs,parse_constant=invalid)
 if not isinstance(v,dict) or set(v)!={'model','stream','input','max_output_tokens','truncation','store','reasoning'}:raise ValueError('top shape')
 if v['model']!='deepseek-flash' or v['stream'] is not True or v['store'] is not False or v['truncation']!='disabled' or v['reasoning']!={'effort':'none'} or type(v['max_output_tokens']) is not int or v['max_output_tokens']<1:raise ValueError('option')
 if type(v['input']) is not list or not 1<=len(v['input'])<=14:raise ValueError('messages')
 for m in v['input']:
  if not isinstance(m,dict) or set(m)!={'role','content'} or m['role'] not in {'system','user','assistant'} or not isinstance(m['content'],str):raise ValueError('message shape')
 if sum(len(m['content']) for m in v['input'])>12000:raise ValueError('character budget')
 if canon(v)!=raw:raise ValueError('noncanonical')
 return v

def manual(v):
 # Responses convert: only adjacent assistant textual messages are joined.
 # Empty new assistant is ignored after existing assistant; separator only
 # when both texts nonempty. Then encoding joins adjacent user with two LF.
 converted=[]
 for m in v['input']:
  r,t=m['role'],m['content']
  if r=='assistant' and converted and converted[-1][0]=='assistant':
   if t:
    old=converted[-1][1];converted[-1]=('assistant',old+('\n\n' if old else '')+t)
  else:converted.append((r,t))
 normalized=[]
 for r,t in converted:
  if r=='user' and normalized and normalized[-1][0]=='user':normalized[-1]=('user',normalized[-1][1]+'\n\n'+t)
  else:normalized.append((r,t))
 s='<｜begin▁of▁sentence｜>'
 for r,t in normalized:
  if r=='system':s+='<｜System｜>'+t
  elif r=='user':s+='<｜User｜>'+t
  else:s+='<｜Assistant｜></think>'+t+'<｜end▁of▁sentence｜>'
 return s+'<｜Assistant｜></think>'
