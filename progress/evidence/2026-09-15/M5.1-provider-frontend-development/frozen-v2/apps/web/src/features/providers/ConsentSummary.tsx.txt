import type { ConsentProposalView, ConsentView, FrozenOutboundSummary } from '../../../../../packages/contracts/generated/api-types'

export function ConsentSummary({ value }: { value: FrozenOutboundSummary }) {
  const assurance = value.input_token_assurance, estimate = value.cost_estimate
  return <section className="provider-summary" aria-label="冻结外发摘要"><h4>本次服务端冻结范围</h4>
    <dl><dt>目的与原任务</dt><dd>{value.purpose} · {value.job_id} · 原修订 {value.source_job_revision}</dd><dt>提供商与模型</dt><dd>{value.provider_id} · r{value.provider_revision} · {value.model}</dd><dt>目的地</dt><dd>{value.base_url} · {value.endpoint_policy}</dd><dt>适配版本</dt><dd>{value.adapter} · {value.adapter_version}</dd><dt>输入字符</dt><dd>{value.input_character_count} 个 Unicode 码点；按实际发送出现次数计入</dd><dt>输入 token 准入</dt><dd>{assurance.kind === 'local_exact' ? `本地完整精确计数 ${assurance.input_tokens}` : `已核上界 ${assurance.input_tokens_upper_bound}（不是实际用量）`} · {assurance.checker_version}</dd><dt>硬预算</dt><dd>输入 ≤ {value.budget.max_input_tokens}；输出 ≤ {value.budget.max_output_tokens}；提供商调用 {value.budget.max_provider_calls} 次；搜索 {value.budget.max_search_calls} 次；工具 {value.budget.max_tool_calls} 次；总超时 {value.budget.timeout_seconds} 秒</dd><dt>联网范围</dt><dd>不允许网页搜索或工具；仅此冻结提供商请求</dd><dt>费用</dt><dd>{estimate.kind === 'unknown' ? '价格未知，金额没有硬保证' : `本机最大估计 USD ${estimate.maximum_estimated_cost}，不保证实际账单`}{value.budget.max_cost_usd === null ? '；未设置金额上限' : `；申请金额上限 USD ${value.budget.max_cost_usd}`}</dd><dt>有效期</dt><dd>{value.created_at} 至 {value.expires_at}</dd></dl>
    <details><summary>核对消息、来源与冻结哈希</summary><p>仅显示安全定位和摘要，不显示实际正文。</p>{value.messages.map((message, index) => <p key={index}>消息 {index + 1} · {message.role} · {message.character_count} 字符 · <code>{message.content_sha256}</code></p>)}{value.references.length === 0 ? <p>该冻结材料没有引用摘录。</p> : value.references.map((reference, index) => <article key={index}><strong>{reference.title}</strong><p>{reference.locator} · {reference.character_count} 字符</p><p>{reference.ref.entity}/{reference.ref.id} · r{reference.ref.revision} · <code>{reference.ref.sha256}</code></p><p>摘录 <code>{reference.excerpt_sha256}</code></p></article>)}<dl><dt>原任务输入</dt><dd><code>{value.source_input_sha256}</code></dd><dt>准备输入</dt><dd><code>{value.input_sha256}</code></dd><dt>请求字节</dt><dd><code>{assurance.request_body_sha256}</code></dd><dt>计量证明</dt><dd><code>{assurance.proof_sha256}</code></dd><dt>配置</dt><dd><code>{value.config_sha256}</code></dd><dt>上下文</dt><dd>{value.context_snapshot_id} · <code>{value.context_snapshot_sha256}</code></dd></dl></details>
  </section>
}
export function ProposalDisplay({ value }: { value: ConsentProposalView }) {
  return <section aria-label="授权提案"><h3>提案 {value.id}</h3><p role="status">当前诊断：{({ current: '当前有效', stale: '依据已改变', expired: '已过期', unavailable: '当前不可用' })[value.validity]}{value.consent_id ? ` · 已有关联授权 ${value.consent_id}，不能再次批准` : ''}</p>{value.warnings.map((warning, index) => <p key={index}>{warning.message}</p>)}<ConsentSummary value={value.summary} /><p>提案哈希：<code>{value.proposal_sha256}</code></p></section>
}
export function ConsentHistoryEntry({ value }: { value: ConsentView }) {
  const dispatch = value.dispatch, cost = dispatch?.usage.cost
  return <section aria-label={`授权 ${value.id}`}><h4>{value.id} · r{value.revision}</h4><p>当前授权：{({ active: '有效', revoked: '已撤销', expired: '已过期' })[value.status]} · 到期 {value.expires_at}{value.revoked_at ? ` · 实际撤销 ${value.revoked_at}` : ''}</p><details><summary>查看此授权冻结范围</summary><ConsentSummary value={value.summary} /></details>
    {dispatch ? <div aria-label="派发记录"><p>派发 {dispatch.id} · 源任务当前 {dispatch.job.status} · {dispatch.error_code ?? (dispatch.finished_at ? '提供商步骤已结束；不代表源任务已完成' : '提供商步骤未终结')}</p><p>开始 {dispatch.started_at ?? '未知'}；结束 {dispatch.finished_at ?? '未终结'}。</p><p>保守消耗调用 {dispatch.usage.consumed_provider_calls}；真实输入用量 {dispatch.usage.input_tokens ?? '未知'}；真实输出用量 {dispatch.usage.output_tokens ?? '未知'}；耗时 {dispatch.usage.elapsed_ms === null ? '未知' : `${dispatch.usage.elapsed_ms} ms`}。</p><p>费用：{!cost || cost.kind === 'unknown' ? '未知' : cost.kind === 'actual' ? `提供商报告 USD ${cost.amount}` : `本机估计 USD ${cost.amount}，不是账单`}。</p></div> : <p>尚无派发记录。授权成功不表示已经调用模型。</p>}
  </section>
}
