import type { ContentRef, Route, RouteStep } from '../../../../../packages/contracts/generated/types'
import { refKey } from '../reader/target'
export type RouteChoice = { ref: ContentRef; title: string; label: string }
export function RouteEditorFields({ value, choices, disabled, change }: { value: Route; choices: RouteChoice[]; disabled: boolean; change: (value: Route) => void }) {
  const update = (id: string, patch: Partial<RouteStep>) => change({ ...value, steps: value.steps.map(step => step.id === id ? { ...step, ...patch } : step) })
  const move = (index: number, delta: number) => { const steps = [...value.steps]; [steps[index], steps[index + delta]] = [steps[index + delta], steps[index]]; change({ ...value, steps }) }
  return <fieldset disabled={disabled}><legend>路线内容</legend><label>路线名称<input aria-label="路线名称" value={value.title} onChange={event => change({ ...value, title: event.target.value })} /></label><label>学习目标<textarea aria-label="路线学习目标" value={value.goal} onChange={event => change({ ...value, goal: event.target.value })} rows={3} /></label><label>添加真实学习任务<select aria-label="添加学习任务" value="" onChange={event => { const item = choices.find(choice => refKey(choice.ref) === event.target.value); if (item) change({ ...value, steps: [...value.steps, { id: `step_${crypto.randomUUID()}`, title: item.title, target: item.ref, requires_steps: [], completion_rule: item.ref.entity === 'practice_set' ? 'practice_submitted' : item.ref.entity === 'assessment' ? 'assessment_submitted' : 'read' }] }) }}><option value="">选择精确课程内容、习题或测试</option>{choices.map(choice => <option key={refKey(choice.ref)} value={refKey(choice.ref)}>{choice.label}</option>)}</select></label>{!choices.length && <p>尚无可添加的真实材料。可以先导入教材，再创建路线；未编造任务目标。</p>}
    <ol className="route-editor-steps">{value.steps.map((step, index) => <li key={step.id}><label>任务 {index + 1} 标题<input aria-label={`任务 ${index + 1} 标题`} value={step.title} onChange={event => update(step.id, { title: event.target.value })} /></label><p className="learning-muted">{step.target.entity} · {step.target.id} · r{step.target.revision}</p><label>完成规则<select aria-label={`任务 ${index + 1} 完成规则`} value={step.completion_rule} onChange={event => update(step.id, { completion_rule: event.target.value as RouteStep['completion_rule'] })}><option value="manual">仅人工标记</option>{['lesson', 'block'].includes(step.target.entity) && <option value="read">明确标记阅读完成</option>}{step.target.entity === 'practice_set' && <option value="practice_submitted">提交本次习题练习</option>}{step.target.entity === 'assessment' && <option value="assessment_submitted">提交本次测试</option>}</select></label><details><summary>先修提醒（不锁定学习入口）</summary>{value.steps.filter(other => other.id !== step.id).map(other => <label key={other.id} className="learning-check"><input type="checkbox" checked={(step.requires_steps ?? []).includes(other.id)} onChange={event => update(step.id, { requires_steps: event.target.checked ? [...(step.requires_steps ?? []), other.id] : (step.requires_steps ?? []).filter(id => id !== other.id) })} />{other.title || '未命名任务'}</label>)}</details><div className="learning-actions"><button type="button" disabled={disabled || index === 0} onClick={() => move(index, -1)} aria-label={`上移任务 ${index + 1}`}>上移</button><button type="button" disabled={disabled || index === value.steps.length - 1} onClick={() => move(index, 1)} aria-label={`下移任务 ${index + 1}`}>下移</button><button type="button" onClick={() => change({ ...value, steps: value.steps.filter(other => other.id !== step.id) })} aria-label={`删除任务 ${index + 1}`}>删除任务</button></div></li>)}</ol>
  </fieldset>
}
