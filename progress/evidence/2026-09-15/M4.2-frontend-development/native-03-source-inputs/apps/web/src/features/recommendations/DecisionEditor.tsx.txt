import { decisionLabels } from './recommendationModel'
import { RecommendationRef } from './RecommendationSources'
import type { useDecisions } from './useDecisions'
export function DecisionEditor({ state }: { state: ReturnType<typeof useDecisions> }) {
  const value = state.editor
  if (!value) return null
  const disabled = state.busy || state.journal.saving || !state.safe || state.competing.length > 0
  return <section className="recommendation-editor" aria-label="推荐决定编辑"><h3>{value.base.target_title} · 记录选择</h3><p>接受只记录你的选择。打开内容是另一个操作；不会自动完成路线、改变能力记录或联网。</p><details><summary>核对原精确目标</summary><RecommendationRef value={value.base.target_ref} /></details><p role="status">{value.acknowledged ? `原命令已确认 · 决定修订 ${value.acknowledged.revision} · ${value.acknowledged.applied ? '已追加决定历史' : '选择与理由相同，未增加修订'}` : '此决定尚未确认保存到服务端'}。</p>
    <label>推荐决定<select value={value.fields.decision} disabled={disabled || state.sealed || !!value.acknowledged} onChange={event => state.edit({ ...value.fields, decision: event.target.value === 'accepted' ? 'accepted' : 'dismissed' })}><option value="accepted">接受</option><option value="dismissed">拒绝</option></select></label><label>决定理由（可不填）<textarea value={value.fields.reason ?? ''} rows={3} disabled={disabled || state.sealed || !!value.acknowledged} onChange={event => state.edit({ ...value.fields, reason: event.target.value || null })} /></label>
    {(state.conflict || state.rejected || !!value.acknowledged) && <section className="learning-comparison" aria-label="推荐决定三方比较"><h4>原基准、本页候选与服务端</h4><div><section><h5>原始基准 · r{value.base.decision_revision}</h5><p>{decisionLabels[value.base.decision]}</p><p>{value.base.decision_reason ?? '无理由'}</p><code>{value.base.decision_sha256}</code></section><section><h5>本页候选</h5><p>{decisionLabels[value.fields.decision]}</p><p>{value.fields.reason ?? '无理由'}</p><p>{value.acknowledged ? `原回执 r${value.acknowledged.revision}` : '原命令未确认'}</p></section><section><h5>当前服务端{state.remote ? ` · r${state.remote.decision_revision}` : ''}</h5><p>{state.remote ? decisionLabels[state.remote.decision] : '尚未读回'}</p><p>{state.remote?.decision_reason ?? '无已读回理由'}</p>{state.remote && <code>{state.remote.decision_sha256}</code>}</section></div><p>{state.rejected ? '原命令因服务端版本冲突被拒绝；比较保持打开。更正需要明确采用已读回基准并再次保存。' : value.acknowledged ? '原命令回执与当前服务端决定分别展示；其他页面后来的更正不会改变原回执。' : '读回结果不替代原命令回执。请先用原 key 重试，核对命令是否曾经成功。'}</p></section>}
    {state.remote?.staleness === 'stale' && <p>原推荐依据已过期。原命令仍可核对回执；不会把此选择移到新推荐。</p>}
    <div className="learning-actions">{!value.acknowledged && <button disabled={disabled || state.rejected} onClick={() => void state.save()}>{state.sealed ? '重试原推荐决定命令' : '保存推荐决定'}</button>}{(state.rejected || !!value.acknowledged) && <button disabled={disabled || !state.remote || state.remote.staleness !== 'current'} onClick={state.correct}>采用服务端基准，显式更正决定</button>}<button disabled={state.busy} onClick={() => void state.refresh()}>重新读取当前决定</button><button disabled={disabled} onClick={state.retain}>{value.acknowledged ? '返回推荐列表' : '保留本机候选并返回列表'}</button>{state.journal.error && <button onClick={state.retryLocal}>重试本机推荐保存</button>}</div>
  </section>
}
