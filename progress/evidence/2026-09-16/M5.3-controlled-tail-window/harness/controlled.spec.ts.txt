import { expect, test, type Page } from '<private-home>/Desktop/learning/Learning_Workbench/apps/web/node_modules/@playwright/test/index.mjs'
import { writeFileSync } from 'node:fs'
import type { AttemptSnapshot, ProviderConfigWrite, SessionResponse, TutorMessagePage, TutorRunCreate, TutorRunView, TutorThreadView } from '<private-home>/.cache/learning-workbench-acceptance/m53-active/packages/contracts/generated/api-types'
import { RestartRuntime } from '<private-home>/.cache/learning-workbench-acceptance/m53-active/tests/e2e/restartRuntime'
import { TutorRuntime } from '<private-home>/.cache/learning-workbench-acceptance/m53-active/tests/e2e/tutorRuntime'
import { observeTutorCompletion } from '<private-home>/.cache/learning-workbench-acceptance/m53-active/tests/e2e/tutorDiagnostic'
import { importAssessmentPackage, originalAssessmentPackage } from '<private-home>/.cache/learning-workbench-acceptance/m53-active/tests/e2e/assessmentTestData'

import { performance } from 'node:perf_hooks'
import { createHash } from 'node:crypto'
test('original third Tutor case with real completed response delivery phase', async ({ playwright }, info) => {
  test.setTimeout(90_000)
  const runtime = await TutorRuntime.start(), errors: string[] = []
  let diagnostic: Awaited<ReturnType<typeof observeTutorCompletion>> | undefined
  const phaseText = process.env.TUTOR_COMPLETION_PHASE_MS ?? 'none'
  if (!['none', '4750', '4600', '4800'].includes(phaseText)) throw new Error('UNDECLARED_PHASE')
  const phase = phaseText === 'none' ? null : Number(phaseText)
  const events: object[] = [], probes: Promise<unknown>[] = []
  const mark = (kind: string, value: object = {}) => events.push({ kind, node_ms: performance.now(), ...value })
  let assertionStarted: number | null = null, verdict = 'NOT_REACHED', gated = false, completeTail = false
  let completedDOMObserved = false
  try {
    const context = await runtime.openBrowser(playwright.chromium), page = context.pages()[0]
    diagnostic = await observeTutorCompletion(page, { readRuntime: () => runtime.diagnosticControl() })
    const exactCompleted = page.getByRole('region', { name: '真实问答线程与任务', exact: true }).getByRole('heading', { name: '真实任务状态：completed', exact: true })
    await page.exposeBinding('__tutorTailWindowDOM', (_source, state: object) => {
      mark('dom_completed_receipt', { state })
      if (completedDOMObserved) return
      completedDOMObserved = true
      mark('exact_role_probe_started')
      probes.push(exactCompleted.evaluateAll(nodes => ({ match_count: nodes.length, nodes: nodes.map(node => {
        const element = node as HTMLElement, box = element.getBoundingClientRect(), style = getComputedStyle(element)
        return { hidden_attribute_ancestor: !!element.closest('[hidden]'), aria_hidden_ancestor: !!element.closest('[aria-hidden="true"]'),
          css_visibility_visible: style.visibility === 'visible', css_display_not_none: style.display !== 'none',
          positive_geometry: box.width > 0 && box.height > 0,
          width: box.width, height: box.height }
      }) })).then(value => mark('exact_role_probe_finished', { value })).catch(() => mark('exact_role_probe_failed')))
    })
    const installTailObserver = () => {
      let emitted = false
      const inspect = () => {
        const regions = [...document.querySelectorAll('[aria-label="真实问答线程与任务"]')]
        const nodes = regions.flatMap(region => [...region.querySelectorAll('[aria-label="当前问答任务"] h3')]).filter(node => node.textContent === '真实任务状态：completed')
        if (!nodes.length || emitted) return
        emitted = true
        const state = { matching_region_count: regions.length, completed_text_count: nodes.length, nodes: nodes.map(node => {
          const element = node as HTMLElement, box = element.getBoundingClientRect(), style = getComputedStyle(element)
          return { hidden_attribute_ancestor: !!element.closest('[hidden]'), aria_hidden_ancestor: !!element.closest('[aria-hidden="true"]'),
            css_visibility_visible: style.visibility === 'visible', css_display_not_none: style.display !== 'none',
            positive_geometry: box.width > 0 && box.height > 0, width: box.width, height: box.height }
        }) }
        void (window as unknown as { __tutorTailWindowDOM: (state: object) => Promise<void> }).__tutorTailWindowDOM(state).catch(() => undefined)
      }
      new MutationObserver(inspect).observe(document, { subtree: true, childList: true, characterData: true, attributes: true, attributeFilter: ['hidden','aria-hidden','style','class'] })
      inspect()
    }
    await page.addInitScript(installTailObserver)
    await page.evaluate(installTailObserver)
    page.on('pageerror', error => errors.push(error.message))
    await runtime.authenticateOnly(page)
    await expect(page.locator('.import-trigger')).toBeEnabled()
    const fixture = originalAssessmentPackage('tutornativeapproved')
    const imported = await importAssessmentPackage(page, fixture)
    await imported.dialog.getByRole('button', { name: '关闭导入', exact: true }).click()
    const initialControl = runtime.control()
    expect(initialControl.test_only).toBe(true); expect(initialControl.received_request_count).toBe(0)
    const session: SessionResponse = await page.request.get('/api/v1/session').then(value => value.json())
    const headers = { Origin: runtime.origin, 'X-CSRF-Token': session.csrf_token }
    const config: ProviderConfigWrite = { expected_revision: 0, adapter: initialControl.adapter, base_url: initialControl.base_url, model: initialControl.model, embedding_model: null, endpoint_policy: 'explicit_loopback', pricing: null }
    expect((await page.request.put('/api/v1/providers/provider_tutor_loopback/config', { data: config, headers: { ...headers, 'Idempotency-Key': 'synthetic-loopback-config' } })).status()).toBe(200)
    expect((await page.request.post('/api/v1/providers/provider_tutor_loopback/secret', { data: { expected_revision: 1, secret: 'synthetic-loopback-native-constant' }, headers: { ...headers, 'Idempotency-Key': 'synthetic-loopback-secret' } })).status()).toBe(200)
    await page.goto(`${runtime.origin}/?reader=${encodeURIComponent(JSON.stringify({ course: fixture.course, lesson: fixture.lesson }))}`)
    await expect(page.locator('.real-reader > h1')).toBeVisible()
    const tutor = page.getByRole('region', { name: '真实问答线程与任务', exact: true })
    const creating = page.waitForResponse(value => value.request().method() === 'POST' && value.url().endsWith('/api/v1/threads'))
    await tutor.getByRole('button', { name: '明确创建本地线程', exact: true }).click()
    const thread: TutorThreadView = await (await creating).json()
    await page.getByLabel('问题草稿', { exact: true }).fill('请解释原创材料，并保留未经验证的边界。')
    await expect(page.getByRole('button', { name: '创建本次问答任务 ↑', exact: true })).toBeEnabled()
    const starting = page.waitForResponse(value => value.request().method() === 'POST' && value.url().endsWith('/api/v1/tutor/runs'))
    await page.getByRole('button', { name: '创建本次问答任务 ↑', exact: true }).click()
    const ack: TutorRunView = await (await starting).json()
    diagnostic.bindRun(ack.run.id)
    await expect(tutor.getByRole('heading', { name: '真实任务状态：awaiting_approval', exact: true })).toBeVisible()
    expect(runtime.control().received_request_count).toBe(0)
    const preview = tutor.getByRole('region', { name: '准备授权预览', exact: true })
    await preview.getByLabel('最大输入 token', { exact: true }).fill('20000')
    await preview.getByLabel('最大输出 token', { exact: true }).fill('2000')
    await preview.getByLabel('到期时间 UTC', { exact: true }).fill(new Date(Date.now() + 600_000).toISOString())
    await preview.getByRole('button', { name: '准备服务端预览命令', exact: true }).click()
    await tutor.getByRole('button', { name: '确认发送授权预览', exact: true }).click()
    await expect(tutor.getByLabel('我已核对本次冻结范围、目的地、预算与到期时间', { exact: true })).toBeVisible()
    expect(runtime.control().received_request_count).toBe(0)
    await tutor.getByLabel('我已核对本次冻结范围、目的地、预算与到期时间', { exact: true }).check()
    await tutor.getByRole('button', { name: '准备批准本次授权命令', exact: true }).click()
    await page.route(`**/api/v1/runs/${ack.run.id}`, async route => {
      if (route.request().method() !== 'GET') { await route.continue(); return }
      const response = await route.fetch(), bytes = await response.body()
      const current: TutorRunView = JSON.parse(bytes.toString('utf8'))
      const sha = createHash('sha256').update(bytes).digest('hex')
      mark('real_run_response_received', { id: current.run?.id, status: current.run?.status, http_status: response.status(),
        last_seq: current.run?.last_seq, job_revision: current.job_revision, body_sha256: sha, body_bytes: bytes.length })
      if (!gated && phase !== null && assertionStarted !== null && response.status() === 200 && current.run.id === ack.run.id && current.run.status === 'completed') {
        gated = true
        const releaseAt = assertionStarted + phase
        mark('hold_real_completed_response', { scheduled_release_ms: releaseAt, response_sha256: sha, delay_ms: Math.max(0, releaseAt - performance.now()) })
        await new Promise(resolve => setTimeout(resolve, Math.max(0, releaseAt - performance.now())))
        mark('release_original_completed_bytes', { response_sha256: sha, release_since_assertion_ms: performance.now() - assertionStarted })
      }
      // This is the exact received body; status/headers come from the same actual APIResponse.
      await route.fulfill({ response, body: bytes })
      mark('original_run_response_fulfilled', { response_sha256: sha, status: current.run?.status })
    })
    diagnostic.mark('grant_click')
    await tutor.getByRole('button', { name: '确认发送批准授权', exact: true }).click()
    await diagnostic.around(info, async () => {
      assertionStarted = performance.now(); mark('original_expect_started', { assertion_started_ms: assertionStarted })
      try {
        await expect(tutor.getByRole('heading', { name: '真实任务状态：completed', exact: true })).toBeVisible()
        verdict = 'passed'; mark('original_expect_passed')
      } catch (error) { verdict = 'failed'; mark('original_expect_failed'); throw error }
    })
    const complete: TutorRunView = await page.request.get(`/api/v1/runs/${ack.run.id}`).then(value => value.json())
    expect(complete.run.id).toBe(ack.run.id); expect(complete.run.thread_id).toBe(thread.id)
    expect(complete.consent_id).toBeTruthy(); expect(complete.latest_proposal_id).toBeTruthy()
    expect(complete.run.answer_markdown).toBe(initialControl.answer_markdown)
    expect(complete.run.citations).toEqual([]); expect(complete.result.provider?.outcome).toBe('complete')
    expect(complete.result.usage.input_tokens).toBeGreaterThan(0); expect(complete.result.usage.output_tokens).toBeGreaterThan(0)
    await expect.poll(() => runtime.control().validated_request_count).toBe(1)
    expect(runtime.control().received_request_count).toBe(1); expect(runtime.control().invalid_request_count).toBe(0)
    await expect(tutor.getByLabel('本次模型回答原文', { exact: true })).toHaveText(initialControl.answer_markdown)
    await expect(tutor.getByText('这些记录仅证明输入了什么，不证明答案或推导受到这些材料支持。', { exact: true })).toBeVisible()
    await expect(tutor.getByText('尚未外部搜索，没有受检外部来源；模型自报链接不作为已核引文。', { exact: true })).toBeVisible()
    await page.reload()
    await tutor.getByRole('button', { name: `${thread.title} · r${complete.thread_revision}`, exact: true }).click()
    await expect(tutor.getByRole('heading', { name: '真实任务状态：completed', exact: true })).toBeVisible()
    await expect(tutor.getByLabel('本次模型回答原文', { exact: true })).toHaveText(initialControl.answer_markdown)
    expect(runtime.control().received_request_count).toBe(1)
    await tutor.getByLabel('本次模型回答原文', { exact: true }).scrollIntoViewIfNeeded()
    await page.screenshot({ path: info.outputPath('tutor-loopback-answer-1440.png') })
    await page.setViewportSize({ width: 390, height: 844 })
    await page.getByRole('button', { name: '切换 Agent 栏', exact: true }).click()
    await expect(page.getByRole('dialog', { name: 'Agent 助教', exact: true })).toBeVisible()
    await tutor.getByRole('button', { name: `${thread.title} · r${complete.thread_revision}`, exact: true }).click()
    await expect(tutor.getByLabel('本次模型回答原文', { exact: true })).toHaveText(initialControl.answer_markdown)
    await tutor.getByLabel('本次模型回答原文', { exact: true }).scrollIntoViewIfNeeded()
    await page.screenshot({ path: info.outputPath('tutor-loopback-answer-390.png') })
    expect(errors).toEqual([])
    writeFileSync(info.outputPath('actual-tutor-loopback.json'), JSON.stringify({ scope: 'Test-only complete-byte model, synthetic secret and original material; actual loopback HTTP, provider ledger, Tutor worker/SSE and native UI. No production input proof, vendor call or answer-quality acceptance.', thread, create_ack: ack, final: complete, observation: runtime.control(), runtime_errors: errors }, null, 2))
    completeTail = true
  } finally {
    try { writeFileSync(info.outputPath('completed-response-phase.json'), JSON.stringify({
      version: 1, phase_ms: phase, clock: 'All node_ms are the same Node performance.now clock in this test worker; browser clocks are not collected.',
      assertion_started_ms: assertionStarted, original_expect_timeout_ms: 5000, original_expect_verdict: verdict,
      original_business_tail_completed: completeTail, gated, events,
      scope: 'Only real same-Run completed GET bytes may be held; queued responses are unchanged. Exact role probe is an additional read and may perturb scheduling. Its start/finish differ from DOM notification time. No expected FAIL verdict; late DOM after5000ms is not a tail-window reproduction. No request/response text, credentials or headers stored.',
    }, null, 2)) } catch { console.error('Private phase metadata incomplete; original case verdict is unchanged.') }
    diagnostic?.dispose(); await runtime.close()
  }
})
