MIGRATION = '0019_draft_edits.sql'


def owners(database):
    imports = ImportService(database)
    candidates = DraftCandidates({'import': imports})
    reviews = ReviewService(database, candidates, ReviewNumeric(candidates, {}),
        {(profile, 'import'): imports for profile in IMPORT_ARTIFACT_PROFILES})
    return imports, reviews, DraftPublicationService(database, reviews, imports)


@pytest.fixture
def previous(tmp_path):
    settings = Settings(data_dir=tmp_path / 'data')
    prior = tmp_path / 'prior'
    prior.mkdir()
    for path in settings.migrations_dir.glob('*.sql'):
        if path.name < '0019':
            shutil.copyfile(path, prior / path.name)
    database = Database(replace(settings, migrations_dir=prior))
    database.initialize()
    _, learner = consume_bootstrap(database, issue_bootstrap_code(database))
    SessionService(database).switch_role(learner, RoleRequest(role='author'), 'role')
    identity = replace(learner, role='author')
    imports, reviews, publication = owners(database)
    worker = ImportWorker(database)
    try:
        staged = imports.stage(identity, data=b'Synthetic source for local editing.\n', kind='markdown',
            filename='synthetic.md', key='source')
        assert worker.run_once()
    finally:
        worker.stop()
    draft = next(d for d in (imports.draft(identity, item) for item in imports.preview(identity, staged.import_id).preview_refs)
                 if d.kind == 'block')
    candidate = dm.DraftCandidate(draft_id=draft.id, draft_revision=1, entity='block', candidate_sha256=draft.candidate_sha256)
    receipt = reviewed(database, identity, candidate, reviews, math='NOT_APPLICABLE')
    request = publish_request(database, identity, candidate, reviews, receipt)
    ref = publication.publish(identity, candidate.draft_id, request, 'publication')
    return database, settings, identity, candidate, receipt, request, ref


def test_real_review_publication_history_survives_edit_owner_schema(previous):
    old, settings, identity, candidate, receipt, request, ref = previous
    before = table_hashes(old)
    migrated_tables = ['draft_candidate_identities', 'draft_candidate_revisions', 'reviews', 'review_jobs',
        'review_revisions', 'review_artifact_bindings', 'review_commands', 'draft_publications',
        'draft_publication_events', 'draft_publication_results', 'draft_publication_commands']
    with old.connect() as conn:
        original_rows = {name: [tuple(row) for row in conn.execute(f'SELECT rowid,* FROM {name} ORDER BY rowid')]
                         for name in migrated_tables}
    upgraded = Database(settings)
    upgraded.initialize()
    after = table_hashes(upgraded)
    assert {k: after[k] for k in before if k != 'schema_migrations'} == {k: v for k, v in before.items() if k != 'schema_migrations'}
    with upgraded.connect() as conn:
        assert {name: [tuple(row) for row in conn.execute(f'SELECT rowid,* FROM {name} ORDER BY rowid')]
                for name in migrated_tables} == original_rows
    imports, reviews, publication = owners(upgraded)
    assert reviews.read(identity, receipt.id) == receipt
    assert publication.publish(identity, candidate.draft_id, request, 'publication') == ref
    with upgraded.transaction() as conn:
        assert conn.execute('PRAGMA foreign_key_check').fetchall() == []
        conn.execute("INSERT INTO draft_candidate_identities VALUES('draft_synthetic_edit',?,'authoring','authoring_edit','block')",
                     (identity.workspace_id,))
        assert conn.execute("SELECT name FROM sqlite_master WHERE name='draft_edit_versions'").fetchone()
    upgraded.initialize()
    assert list((settings.data_dir / 'backups').glob('*.sqlite3'))


@pytest.mark.parametrize('failure', ['after_schema', 'after_history'])
def test_forward_failure_restores_original_schema_and_history(previous, tmp_path, failure):
    old, settings, identity, candidate, receipt, request, ref = previous
    before = table_hashes(old)
    with old.connect() as conn:
        original_schema = [tuple(r) for r in conn.execute('SELECT type,name,tbl_name,sql FROM sqlite_master ORDER BY type,name')]
    broken = tmp_path / 'broken'
    shutil.copytree(settings.migrations_dir, broken)
    target = broken / MIGRATION
    text = target.read_text()
    marker = '-- EDIT_SCHEMA_REBUILT' if failure == 'after_schema' else '-- EDIT_HISTORY_RESTORED'
    assert marker in text
    target.write_text(text.replace(marker, 'SELECT * FROM synthetic_missing_migration_table;\n' + marker))
    with pytest.raises(sqlite3.Error):
        Database(replace(settings, migrations_dir=broken)).initialize()
    assert table_hashes(old) == before
    with old.connect() as conn:
        assert [tuple(r) for r in conn.execute('SELECT type,name,tbl_name,sql FROM sqlite_master ORDER BY type,name')] == original_schema
        assert conn.execute('PRAGMA foreign_key_check').fetchall() == []
    _, reviews, publication = owners(old)
    assert reviews.read(identity, receipt.id) == receipt
    assert publication.publish(identity, candidate.draft_id, request, 'publication') == ref
