import { useEffect, useRef, useState, useSyncExternalStore } from 'react'
import type { ContentRef, RevisionSummary } from '../../../../../../packages/contracts/generated/api-types'
import { getSessionGeneration, subscribeSessionAccess } from '../../../api/client'
import type { LoadedBlock } from '../contentClient'
import type { ComparePort } from './compareClient'

type State = { key: string; access: number; open: boolean; items: RevisionSummary[]; cursor: string | null; left: ContentRef | null; right: ContentRef | null; result: [LoadedBlock, LoadedBlock] | null; busy: boolean; error: string }
const empty = (key: string, access: number): State => ({ key, access, open: false, items: [], cursor: null, left: null, right: null, result: null, busy: false, error: '' })
export function useVersionCompare(workspace: string | null, id: string, port: ComparePort) {
  const access = useSyncExternalStore(subscribeSessionAccess, getSessionGeneration, getSessionGeneration)
  const key = JSON.stringify([workspace, id])
  const owner = useRef({ key, access, sequence: 0 })
  if (owner.current.key !== key || owner.current.access !== access) owner.current = { key, access, sequence: owner.current.sequence + 1 }
  const [stored, setStored] = useState(() => empty(key, access))
  const state = stored.key === key && stored.access === access ? stored : empty(key, access)
  const patch = (changes: Partial<State>) => setStored(old => ({ ...(old.key === key && old.access === access ? old : empty(key, access)), ...changes }))
  useEffect(() => () => { owner.current.sequence++ }, [])
  useEffect(() => { setStored(old => old.key === key && old.access === access ? old : empty(key, access)) }, [key, access])
  useEffect(() => {
    const clear = () => { owner.current.sequence++; patch({ result: null, busy: false, error: '读取权限需要重新核对，请明确重新读取所选修订。' }) }
    const visibility = () => { if (document.visibilityState !== 'visible') clear() }
    window.addEventListener('focus', clear); document.addEventListener('visibilitychange', visibility)
    return () => { window.removeEventListener('focus', clear); document.removeEventListener('visibilitychange', visibility) }
  }, [key, access])
  const begin = () => {
    const sequence = ++owner.current.sequence
    patch({ busy: true, error: '', result: null })
    return () => owner.current.key === key && owner.current.access === access && owner.current.sequence === sequence && getSessionGeneration() === access
  }
  const permitted = async () => {
    const session = await port.session()
    if (!workspace || session.workspace_id !== workspace || session.active_independent_attempt_id !== null) throw new Error('当前工作区或测试策略不允许读取比较内容。')
  }
  const history = async (more = false) => {
    if (!workspace || more && (!state.cursor || state.items.length >= 200)) return
    const valid = begin(), cursor = more ? state.cursor : null
    try {
      await permitted(); if (!valid()) return
      const page = await port.history(id, cursor)
      if (!valid()) return
      await permitted(); if (!valid()) return
      if (more && (page.next_cursor === cursor || page.items.some(item => item.ref.revision >= state.items.at(-1)!.ref.revision))) throw new Error('历史分页重复或冲突，请重新读取历史列表。')
      patch({ items: more ? [...state.items, ...page.items] : page.items, cursor: page.next_cursor })
    } catch (error) { if (valid()) patch({ error: (error as Error).message, result: null }) }
    finally { if (valid()) patch({ busy: false }) }
  }
  const show = () => { patch({ open: true }); if (!state.items.length) void history() }
  const close = () => { owner.current.sequence++; patch({ open: false, busy: false, result: null, error: '' }) }
  const choose = (side: 'left' | 'right', sha: string) => {
    owner.current.sequence++
    patch({ [side]: state.items.find(item => item.ref.sha256 === sha)?.ref ?? null, result: null, busy: false, error: '' })
  }
  const read = async () => {
    if (!state.left || !state.right || state.left.sha256 === state.right.sha256) return
    const left = state.left, right = state.right, valid = begin()
    try {
      await permitted(); if (!valid()) return
      const pair = await Promise.all([port.block(left), port.block(right)])
      if (!valid()) return
      await permitted(); if (!valid()) return
      patch({ result: pair })
    } catch (error) { if (valid()) patch({ error: (error as Error).message, result: null }) }
    finally { if (valid()) patch({ busy: false }) }
  }
  return { ...state, show, close, choose, history, read }
}
