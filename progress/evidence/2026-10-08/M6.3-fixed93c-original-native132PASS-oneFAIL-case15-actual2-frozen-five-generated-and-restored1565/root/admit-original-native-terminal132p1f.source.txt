def finite(root, manifest, seal, count, expected):
    m = load(root / manifest)
    names = set()
    for x in m['finite_members']:
        assert x['name'] not in names
        names.add(x['name'])
        p = root / x['name']
        assert p.is_file() and not p.is_symlink() and p.resolve().is_relative_to(root.resolve())
        b = p.read_bytes()
        assert (len(b), H(b)) == (x['size'], x['sha256'])
    assert len(names) == count
    assert {str(p.relative_to(root)) for p in root.rglob('*') if p.is_file()} == names | {manifest, seal}
    s = load(root / seal)
    assert H((root / manifest).read_bytes()) == expected
    assert s['manifest'] == {'size': (root / manifest).stat().st_size, 'sha256': expected}
    return m
